### Задача 21: Сервис агрегации метрик

Найдите все критические ошибки, race conditions, утечки ресурсов и неэффективные аллокации памяти в данном компоненте. Предложите варианты исправления.

```go
package main

import (
	"fmt"
	"time"
)

type Metric struct {
	Name  string
	Value float64
}

type MetricsAggrerator struct {
	buffer []Metric
	ch     chan Metric
	db     *ClickHouseClient
}

func NewMetricsAggrerator(db *ClickHouseClient) *MetricsAggrerator {
	ma := &MetricsAggrerator{
		buffer: make([]Metric, 0),
		ch:     make(chan Metric),
		db:     db,
	}
	go ma.startAggregation()
	return ma
}

func (ma *MetricsAggrerator) AddMetric(m Metric) {
	ma.ch <- m
}

func (ma *MetricsAggrerator) startAggregation() {
	ticker := time.NewTicker(1 * time.Second)
	for {
		select {
		case metric := <-ma.ch:
			ma.buffer = append(ma.buffer, metric)
			if len(ma.buffer) >= 100 {
				go ma.db.Save(ma.buffer)
				ma.buffer = make([]Metric, 0)
			}
		case <-ticker.C:
			if len(ma.buffer) > 0 {
				go ma.db.Save(ma.buffer)
				ma.buffer = make([]Metric, 0)
			}
		}
	}
}

type ClickHouseClient struct{}

func (db *ClickHouseClient) Save(metrics []Metric) {
	time.Sleep(500 * time.Millisecond)
	fmt.Printf("Saved %d metrics\n", len(metrics))
}
```

### Задача 22: Параллельный HTTP-воркер с кэшированием

Проанализируйте код параллельного обработчика. Найдите скрытые дедлоки, гонки данных, архитектурные проблемы и уязвимости к высоким нагрузкам.

```go
package main

import (
	"sync"
	"time"
)

type UserProfile struct {
	ID   int
	Name string
}

type UserFetcher struct {
	cache map[int]UserProfile
	mu    sync.Mutex
}

func NewUserFetcher() *UserFetcher {
	return &UserFetcher{
		cache: make(map[int]UserProfile),
	}
}

func (uf *UserFetcher) FetchUsersData(userIDs []int) []UserProfile {
	var wg sync.WaitGroup
	ch := make(chan int)
	results := make([]UserProfile, 0)

	for i := 0; i < 5; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for id := range ch {
				uf.mu.Lock()
				profile, exists := uf.cache[id]
				uf.mu.Unlock()

				if exists {
					results = append(results, profile)
					continue
				}

				fetched := uf.fetchFromRemoteAPI(id)

				uf.mu.Lock()
				uf.cache[id] = fetched
				uf.mu.Unlock()

				results = append(results, fetched)
			}
		}()
	}

	for _, id := range userIDs {
		ch <- id
	}
	close(ch)

	wg.Wait()
	return results
}

func (uf *UserFetcher) fetchFromRemoteAPI(id int) UserProfile {
	time.Sleep(100 * time.Millisecond)
	return UserProfile{ID: id, Name: "User"}
}

```

### На что обратить внимание при решении (Глазами интервьера Яндекса):

1. **Каналы каналов (`chan chan T`)**: Для задачи с сохранением порядка (Задача 26) это классическое и очень элегантное решение. Подумай, как передавать "обещание" (future) результата в выходной поток.
2. **Использование `sync.Cond**`: Для динамических воркер-пулов (Задача 23 и 32) вместо постоянного опроса каналов или `time.Sleep` инженеры платформы ценят умение использовать условные переменные (`sync.Cond`) для пробуждения заснувших горутин.
3. **Пакет `runtime**`: Упомяни вслух, что при паниках (Задача 24) полезно использовать `runtime.Stack`, чтобы получить детальный трейс ошибки именно из упавшей горутины, так как обычный трейс в `main` её не покажет.
